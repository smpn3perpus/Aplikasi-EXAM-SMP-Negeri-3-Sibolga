```typescript // supabase/functions/daftar-akun/index.ts import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!; const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!; const adminClient = createClient( SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, { auth: { autoRefreshToken: false, persistSession: false, }, } ); // ===================================================== // RESPONSE // ===================================================== function jsonResponse( body: any, status = 200 ) { return new Response( JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } // ===================================================== // FUNCTION // ===================================================== Deno.serve(async (req) => { // --------------------------------------------------- // CORS // --------------------------------------------------- if (req.method === "OPTIONS") { return new Response( "ok", { status: 200, headers: corsHeaders, } ); } // --------------------------------------------------- // HANYA POST // --------------------------------------------------- if (req.method !== "POST") { return jsonResponse( { success: false, message: "Method tidak diizinkan.", }, 405 ); } try { // ================================================= // BACA DATA // ================================================= const body = await req.json(); const jenis = String(body.jenis || "") .trim() .toLowerCase(); const identitas = String( body.identitas || body.nip_nik || body.nisn || "" ).trim(); const password = String(body.password || ""); // ================================================= // VALIDASI // ================================================= if ( jenis !== "guru" && jenis !== "siswa" ) { return jsonResponse( { success: false, message: "Jenis akun harus Guru atau Siswa.", }, 400 ); } if (!identitas) { return jsonResponse( { success: false, message: "NIP/NIK atau NISN wajib diisi.", }, 400 ); } if (!password) { return jsonResponse( { success: false, message: "Password wajib diisi.", }, 400 ); } if (password.length < 6) { return jsonResponse( { success: false, message: "Password minimal 6 karakter.", }, 400 ); } // ================================================= // GURU // ================================================= if (jenis === "guru") { const { data: guru, error: guruError, } = await adminClient .from("guru_master") .select( "id,nama,nip_nik,user_id,is_active" ) .eq( "nip_nik", identitas ) .maybeSingle(); if (guruError) { console.error( "guru_master error:", guruError ); return jsonResponse( { success: false, message: "Gagal membaca data guru: " + guruError.message, }, 500 ); } if (!guru) { return jsonResponse( { success: false, message: "NIP/NIK tidak ditemukan di data Guru.", }, 404 ); } // ------------------------------------------------ // JIKA SUDAH TERHUBUNG // ------------------------------------------------ if (guru.user_id) { return jsonResponse( { success: false, message: "Akun Guru dengan NIP/NIK ini sudah dibuat. Silakan login.", }, 409 ); } // ------------------------------------------------ // EMAIL INTERNAL // // Email ini hanya digunakan untuk Supabase Auth. // Guru tetap login menggunakan NIP/NIK. // ------------------------------------------------ const email = `${identitas}@exam.smpn3sibolga.sch.id`; // ------------------------------------------------ // BUAT USER AUTH // ------------------------------------------------ const { data: authData, error: authError, } = await adminClient.auth.admin.createUser({ email: email, password: password, email_confirm: true, user_metadata: { nama: guru.nama, nip_nik: guru.nip_nik, jenis: "guru", }, }); if (authError) { console.error( "Auth Guru error:", authError ); return jsonResponse( { success: false, message: "Gagal membuat akun Guru: " + authError.message, }, 400 ); } const userId = authData.user.id; // ------------------------------------------------ // PROFILE // ------------------------------------------------ const { error: profileError, } = await adminClient .from("profiles") .upsert( { user_id: userId, nama: guru.nama, role: "guru", }, { onConflict: "user_id", } ); if (profileError) { console.error( "Profile Guru error:", profileError ); // Hapus Auth jika profile gagal await adminClient.auth.admin.deleteUser( userId ); return jsonResponse( { success: false, message: "Gagal membuat profile Guru: " + profileError.message, }, 500 ); } // ------------------------------------------------ // TABEL GURU // // Kolom sesuai tabel yang Anda tunjukkan: // id, user_id, nama, nip, username, email // ------------------------------------------------ const { error: guruError2, } = await adminClient .from("guru") .insert({ id: guru.id, user_id: userId, nama: guru.nama, nip: guru.nip_nik, username: guru.nip_nik, email: email, }); if (guruError2) { console.error( "Tabel guru error:", guruError2 ); // Hapus profile await adminClient .from("profiles") .delete() .eq( "user_id", userId ); // Hapus Auth await adminClient.auth.admin.deleteUser( userId ); return jsonResponse( { success: false, message: "Gagal menyimpan data guru: " + guruError2.message, }, 500 ); } // ------------------------------------------------ // HUBUNGKAN guru_master // DAN AKTIFKAN AKUN // ------------------------------------------------ const { error: updateGuruError, } = await adminClient .from("guru_master") .update({ user_id: userId, is_active: true, }) .eq( "id", guru.id ); if (updateGuruError) { console.error( "guru_master update error:", updateGuruError ); // Bersihkan data yang sudah dibuat await adminClient .from("guru") .delete() .eq( "user_id", userId ); await adminClient .from("profiles") .delete() .eq( "user_id", userId ); await adminClient.auth.admin.deleteUser( userId ); return jsonResponse( { success: false, message: "Gagal menghubungkan akun dengan guru_master: " + updateGuruError.message, }, 500 ); } // ------------------------------------------------ // BERHASIL // ------------------------------------------------ return jsonResponse( { success: true, message: "Akun Guru berhasil dibuat dan diaktifkan.", nama: guru.nama, identitas: guru.nip_nik, } ); } // ================================================= // SISWA // ================================================= if (jenis === "siswa") { const { data: siswa, error: siswaError, } = await adminClient .from("siswa_master") .select( "id,nama,nisn,kelas,user_id,is_active" ) .eq( "nisn", identitas ) .maybeSingle(); if (siswaError) { console.error( "siswa_master error:", siswaError ); return jsonResponse( { success: false, message: "Gagal membaca data siswa: " + siswaError.message, }, 500 ); } if (!siswa) { return jsonResponse( { success: false, message: "NISN tidak ditemukan di data Siswa.", }, 404 ); } // ------------------------------------------------ // SUDAH PUNYA AKUN // ------------------------------------------------ if (siswa.user_id) { return jsonResponse( { success: false, message: "Akun Siswa dengan NISN ini sudah dibuat. Silakan login.", }, 409 ); } // ------------------------------------------------ // EMAIL INTERNAL // ------------------------------------------------ const email = `${identitas}@exam.smpn3sibolga.sch.id`; // ------------------------------------------------ // AUTH // ------------------------------------------------ const { data: authData, error: authError, } = await adminClient.auth.admin.createUser({ email: email, password: password, email_confirm: true, user_metadata: { nama: siswa.nama, nisn: siswa.nisn, kelas: siswa.kelas, jenis: "siswa", }, }); if (authError) { console.error( "Auth Siswa error:", authError ); return jsonResponse( { success: false, message: "Gagal membuat akun Siswa: " + authError.message, }, 400 ); } const userId = authData.user.id; // ------------------------------------------------ // PROFILE // ------------------------------------------------ const { error: profileError, } = await adminClient .from("profiles") .upsert( { user_id: userId, nama: siswa.nama, role: "siswa", }, { onConflict: "user_id", } ); if (profileError) { console.error( "Profile Siswa error:", profileError ); await adminClient.auth.admin.deleteUser( userId ); return jsonResponse( { success: false, message: "Gagal membuat profile Siswa: " + profileError.message, }, 500 ); } // ------------------------------------------------ // HUBUNGKAN SISWA_MASTER // ------------------------------------------------ const { error: updateSiswaError, } = await adminClient .from("siswa_master") .update({ user_id: userId, is_active: true, }) .eq( "id", siswa.id ); if (updateSiswaError) { console.error( "siswa_master update error:", updateSiswaError ); await adminClient .from("profiles") .delete() .eq( "user_id", userId ); await adminClient.auth.admin.deleteUser( userId ); return jsonResponse( { success: false, message: "Gagal menghubungkan akun dengan siswa_master: " + updateSiswaError.message, }, 500 ); } // ------------------------------------------------ // BERHASIL // ------------------------------------------------ return jsonResponse( { success: true, message: "Akun Siswa berhasil dibuat dan diaktifkan.", nama: siswa.nama, identitas: siswa.nisn, kelas: siswa.kelas, } ); } return jsonResponse( { success: false, message: "Permintaan tidak valid.", }, 400 ); } catch (error) { console.error( "UNEXPECTED ERROR:", error ); return jsonResponse( { success: false, message: error?.message || "Terjadi kesalahan pada server.", }, 500 ); } }); ```
