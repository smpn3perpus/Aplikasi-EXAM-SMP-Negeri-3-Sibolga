```ts import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const supabaseUrl = Deno.env.get("SUPABASE_URL")!; const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!; const supabase = createClient( supabaseUrl, serviceRoleKey ); Deno.serve(async (req) => { try { if (req.method !== "POST") { return new Response( JSON.stringify({ success: false, error: "Method harus POST" }), { status: 405, headers: { "Content-Type": "application/json" } } ); } const body = await req.json(); const role = String(body.role || "").trim().toLowerCase(); const identifier = String(body.identifier || "").trim(); if (!role || !identifier) { return new Response( JSON.stringify({ success: false, error: "Role dan identifier wajib diisi" }), { status: 400, headers: { "Content-Type": "application/json" } } ); } let table = ""; let column = ""; if (role === "guru") { table = "guru"; column = "nip"; } else if (role === "siswa") { table = "siswa"; column = "nisn"; } else { return new Response( JSON.stringify({ success: false, error: "Role harus guru atau siswa" }), { status: 400, headers: { "Content-Type": "application/json" } } ); } /* * Cari data Guru atau Siswa */ const result = await supabase .from(table) .select("*") .eq(column, identifier) .maybeSingle(); if (result.error) { throw result.error; } const account = result.data; if (!account) { return new Response( JSON.stringify({ success: false, error: "Data pengguna tidak ditemukan" }), { status: 404, headers: { "Content-Type": "application/json" } } ); } /* * Jangan membuat akun kedua */ if (account.user_id) { return new Response( JSON.stringify({ success: true, already_exists: true, message: "Akun sudah terhubung", nama: account.nama, user_id: account.user_id }), { status: 200, headers: { "Content-Type": "application/json" } } ); } /* * Email internal Supabase Auth */ const email = identifier + "@login.smp3.id"; /* * Password awal * Guru = NIP * Siswa = NISN */ const password = identifier; /* * Buat akun Supabase Auth */ const authResult = await supabase.auth.admin.createUser({ email: email, password: password, email_confirm: true, user_metadata: { role: role, nama: account.nama, identifier: identifier } }); if (authResult.error) { return new Response( JSON.stringify({ success: false, error: authResult.error.message }), { status: 400, headers: { "Content-Type": "application/json" } } ); } const authUser = authResult.data.user; if (!authUser) { throw new Error( "User Auth tidak berhasil dibuat" ); } /* * Hubungkan user_id */ const updateResult = await supabase .from(table) .update({ user_id: authUser.id, email: email }) .eq("id", account.id); if (updateResult.error) { await supabase.auth.admin.deleteUser( authUser.id ); throw updateResult.error; } /* * Simpan juga ke profiles */ const profileResult = await supabase .from("profiles") .upsert( { user_id: authUser.id, nama: account.nama, role: role, nisn_nip: identifier, email_login: email }, { onConflict: "user_id" } ); if (profileResult.error) { console.warn( "Profiles tidak berhasil diperbarui:", profileResult.error.message ); } /* * Berhasil */ return new Response( JSON.stringify({ success: true, message: "Akun berhasil dibuat", nama: account.nama, role: role, identifier: identifier, user_id: authUser.id, email: email, password_awal: identifier }), { status: 200, headers: { "Content-Type": "application/json" } } ); } catch (error) { console.error(error); return new Response( JSON.stringify({ success: false, error: error instanceof Error ? error.message : "Terjadi kesalahan" }), { status: 500, headers: { "Content-Type": "application/json" } } ); } }); ```
